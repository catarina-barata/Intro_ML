import pygame import random import numpy as np import sys # Inicialização do Pygame pygame.init() pygame.display.set_caption("Simulador de Reinforcement Learning - Q-Learning")



# Configurações do Grid (Mundo de 5x5) TAMANHO_CELULA = 100 LARGURA_GRID, ALTURA_GRID = 5, 5 LARGURA_TELA = LARGURA_GRID * TAMANHO_CELULA ALTURA_TELA = (ALTURA_GRID * TAMANHO_CELULA) + 60 # Espaço extra para o texto TELA = pygame.display.set_mode((LARGURA_TELA, ALTURA_TELA)) RELOGIO = pygame.time.Clock() # Cores COR_FUNDO = (240, 244, 248) COR_LINHA = (200, 214, 224) COR_AGENTE = (37, 99, 235) # Azul COR_OBJETIVO = (22, 163, 74) # Verde COR_OBSTACULO = (220, 38, 38) # Vermelho COR_TEXTO = (30, 41, 59) # Elementos do Cenário (X, Y) pos_objetivo = (4, 4) pos_obstaculo = (2, 2) # Parâmetros de Reinforcement Learning (Q-Learning) # Ações: 0=Cima, 1=Baixo, 2=Esquerda, 3=Direita acoes = [0, 1, 2, 3] q_table = np.zeros((LARGURA_GRID, ALTURA_GRID, len(acoes))) alpha = 0.1 # Taxa de aprendizagem gamma = 0.9 # Fator de desconto epsilon = 0.3 # Taxa de exploração (30% de passos aleatórios) # Estado inicial do Agente def reiniciar_agente(): while True: x = random.randint(0, LARGURA_GRID - 1) y = random.randint(0, ALTURA_GRID - 1) if (x, y) != pos_objetivo and (x, y) != pos_obstaculo: return [x, y] pos_agente = reiniciar_agente() episodio = 1 pontuacao_total = 0 recompensa_atual = 0 FONTE = pygame.font.SysFont("Arial", 18, bold=True) # Loop Principal do Simulador rodando = True while rodando: TELA.fill(COR_FUNDO) # Processar eventos de fechar janela for evento in pygame.event.get(): if evento.type == pygame.QUIT: rodando = False pygame.quit() sys.exit() # --- LÓGICA DO Q-LEARNING (Tomada de Decisão) --- x_antigo, y_antigo = pos_agente[0], pos_agente[1] # Escolha da ação: Exploração (Aleatória) vs Exploração de Conhecimento (Exploit) if random.uniform(0, 1) < epsilon: acao = random.choice(acoes) # Explorar else: acao = np.argmax(q_table[x_antigo, y_antigo]) # Escolher a melhor conhecida # Executar a ação no mapa if acao == 0 and pos_agente[1] > 0: # Cima pos_agente[1] -= 1 elif acao == 1 and pos_agente[1] < ALTURA_GRID - 1: # Baixo pos_agente[1] += 1 elif acao == 2 and pos_agente[0] > 0: # Esquerda pos_agente[0] -= 1 elif acao == 3 and pos_agente[0] < LARGURA_GRID - 1: # Direita pos_agente[0] += 1 x_novo, y_novo = pos_agente[0], pos_agente[1] # Sistema de Recompensas if (x_novo, y_novo) == pos_objetivo: recompensa = 100 pontuacao_total += 100 pos_agente = reiniciar_agente() # Chegou ao fim, reinicia episodio += 1 elif (x_novo, y_novo) == pos_obstaculo: recompensa = -100 pontuacao_total -= 100 pos_agente = reiniciar_agente() # Bateu, reinicia episodio += 1 else: recompensa = -1 # Penalização por cada passo gasto (para ser rápido) pontuacao_total -= 1 # Atualização da Q-Table (Equação de Bellman simplificada) melhor_q_futuro = np.max(q_table[x_novo, y_novo]) q_table[x_antigo, y_antigo, acao] += alpha * (recompensa + gamma * melhor_q_futuro - q_table[x_antigo, y_antigo, acao]) # --- DESENHO GRÁFICO NA TELA --- # Desenhar as linhas do Grid for x in range(0, LARGURA_TELA, TAMANHO_CELULA): pygame.draw.line(TELA, COR_LINHA, (x, 0), (x, ALTURA_GRID * TAMANHO_CELULA)) for y in range(0, ALTURA_GRID * TAMANHO_CELULA, TAMANHO_CELULA): pygame.draw.line(TELA, COR_LINHA, (0, y), (LARGURA_TELA, y)) # Desenhar Objetivo, Obstáculo e Agente pygame.draw.rect(TELA, COR_OBJETIVO, (pos_objetivo[0]*TAMANHO_CELULA+5, pos_objetivo[1]*TAMANHO_CELULA+5, TAMANHO_CELULA-10, TAMANHO_CELULA-10)) pygame.draw.rect(TELA, COR_OBSTACULO, (pos_obstaculo[0]*TAMANHO_CELULA+5, pos_obstaculo[1]*TAMANHO_CELULA+5, TAMANHO_CELULA-10, TAMANHO_CELULA-10)) pygame.draw.rect(TELA, COR_AGENTE, (pos_agente[0]*TAMANHO_CELULA+15, pos_agente[1]*TAMANHO_CELULA+15, TAMANHO_CELULA-30, TAMANHO_CELULA-30)) # Desenhar o painel de texto inferior (Dashboard de Aula) pygame.draw.rect(TELA, (255, 255, 255), (0, ALTURA_GRID * TAMANHO_CELULA, LARGURA_TELA, 60)) texto_status = FONTE.render(f"Episódio: {episodio} | Pontuação: {pontuacao_total} | Epsilon (Exploração): {epsilon}", True, COR_TEXTO) TELA.blit(texto_status, (15, ALTURA_GRID * TAMANHO_CELULA + 20)) pygame.display.flip() RELOGIO.tick(10) # Controla a velocidade (10 passos por segundo para os alunos verem bem)
